# This renderer's material and SDK are pinned to Metal on Apple Silicon. AUTO
# permits portable 2D builds; release jobs requesting ON must not silently lose
# 3D when their SDK or architecture is wrong.
set(TOON_ENABLE_FILAMENT "AUTO" CACHE STRING "3D reference renderer: AUTO, ON or OFF")
set_property(CACHE TOON_ENABLE_FILAMENT PROPERTY STRINGS AUTO ON OFF)
string(TOUPPER "${TOON_ENABLE_FILAMENT}" _toon_filament_mode)
if(NOT _toon_filament_mode MATCHES "^(AUTO|ON|OFF)$")
    message(FATAL_ERROR "TOON_ENABLE_FILAMENT must be AUTO, ON or OFF")
endif()
set(TOON_FILAMENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/filament/filament" CACHE PATH "Filament SDK")
set(_toon_arch "${CMAKE_SYSTEM_PROCESSOR}")
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
    set(_toon_arch "${CMAKE_OSX_ARCHITECTURES}")
elseif(MSVC AND CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    set(_toon_arch "x86_64")
endif()
set(_toon_filament_available FALSE)
set(_toon_filament_libraries filament backend bluegl bluevk filabridge filaflat utils geometry smol-v ibl abseil zstd)
if(APPLE AND _toon_arch MATCHES "^(arm64|aarch64)$" AND EXISTS "${TOON_FILAMENT_ROOT}/include/filament/Engine.h")
    set(_toon_filament_available TRUE)
    foreach(lib IN LISTS _toon_filament_libraries)
        if(NOT EXISTS "${TOON_FILAMENT_ROOT}/lib/arm64/lib${lib}.a")
            set(_toon_filament_available FALSE)
        endif()
    endforeach()
endif()
if(_toon_filament_mode STREQUAL "ON" AND NOT _toon_filament_available)
    message(FATAL_ERROR "3D was required, but the pinned macOS arm64 Filament SDK is unavailable for this target. Run scripts/setup_filament.sh on Apple Silicon or explicitly choose -DTOON_ENABLE_FILAMENT=OFF for a 2D build.")
endif()
set(TOON_FILAMENT_ENABLED FALSE)
if(NOT _toon_filament_mode STREQUAL "OFF" AND _toon_filament_available)
    set(TOON_FILAMENT_ENABLED TRUE)
    target_compile_definitions(toon_workspace PRIVATE TOON_HAS_FILAMENT=1)
    target_include_directories(toon_workspace SYSTEM PRIVATE "${TOON_FILAMENT_ROOT}/include")
    foreach(lib IN LISTS _toon_filament_libraries)
        target_link_libraries(toon_workspace PRIVATE "${TOON_FILAMENT_ROOT}/lib/arm64/lib${lib}.a")
    endforeach()
    target_link_libraries(toon_workspace PRIVATE "-framework Cocoa" "-framework Metal" "-framework CoreVideo" "-framework QuartzCore")
else()
    message(STATUS "3D reference rendering disabled for this build (${_toon_arch}, ${_toon_filament_mode}). Drawing, animation, audio and project storage remain available.")
endif()

    # Source strings and test fixtures contain Unicode. Propagate encoding to
    # application/test consumers; expose the CRT math constants used by the
    # existing geometry code before any consumer includes <cmath>.
    # MSBuild parallelizes projects; cap the large core/workspace projects at
    # two compiler processes each so a clean Windows build also makes progress.
    target_compile_options(toon_core PUBLIC /utf-8 /bigobj /MP2 PRIVATE /W4)
    target_compile_definitions(toon_core PUBLIC _USE_MATH_DEFINES)
    target_compile_options(toon_workspace PRIVATE /W4)
else()
    target_compile_options(toon_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(toon_workspace PRIVATE -Wall -Wextra -Wpedantic)
endif()

            add_test(NAME ${selection_suite} COMMAND toon_${selection_suite}_tests)
        else()
            add_test(NAME ${selection_suite} COMMAND toon_${selection_suite}_tests -platform offscreen)
        endif()
        set_tests_properties(${selection_suite} PROPERTIES TIMEOUT 60)
    endforeach()
    foreach(ui_suite IN ITEMS ui_theme brush_widgets color_panel guide_storage guide_geometry guide_panel guide_canvas scrub_value)
        qt_add_executable(toon_${ui_suite}_tests tests/test_${ui_suite}.cpp)
        target_link_libraries(toon_${ui_suite}_tests PRIVATE toon_workspace Qt6::Test)
        if(ui_suite STREQUAL "guide_geometry")
            add_test(NAME ${ui_suite} COMMAND toon_${ui_suite}_tests)
        else()
            add_test(NAME ${ui_suite} COMMAND toon_${ui_suite}_tests -platform offscreen)
        endif()
        set_tests_properties(${ui_suite} PROPERTIES TIMEOUT 60)
    endforeach()
    foreach(edit_suite IN ITEMS advanced_transform clipboard_tools)
        qt_add_executable(toon_${edit_suite}_tests tests/test_${edit_suite}.cpp)

        add_test(NAME ${edit_suite} COMMAND toon_${edit_suite}_tests -platform offscreen)
        set_tests_properties(${edit_suite} PROPERTIES TIMEOUT 60)
    endforeach()
    set_tests_properties(foundation drawing controls cursor reference_storage references stabilizer brush_library abr brush_engine brush_panel brush_advanced brush_settings_panel PROPERTIES TIMEOUT 60)
    if(WIN32)
        # QtTest otherwise sends plain output to OutputDebugString on hosted
        # Windows runners, hiding assertion details from CTest and its logs.
        get_property(_toon_registered_tests DIRECTORY PROPERTY TESTS)
        set_property(TEST ${_toon_registered_tests} APPEND PROPERTY ENVIRONMENT "QT_FORCE_STDERR_LOGGING=1")
        # Qt's Windows offscreen plugin uses FreeType rather than the native
        # Windows font database. Without an explicit font directory every
        # glyph uses fallback-box metrics, invalidating drawing/layout checks.
        file(TO_CMAKE_PATH "$ENV{QT_QPA_FONTDIR}" _toon_test_font_dir)
        if(NOT _toon_test_font_dir AND DEFINED ENV{WINDIR})
            file(TO_CMAKE_PATH "$ENV{WINDIR}/Fonts" _toon_test_font_dir)
        endif()
        if(IS_DIRECTORY "${_toon_test_font_dir}")
            set_property(TEST ${_toon_registered_tests} APPEND PROPERTY ENVIRONMENT
                "QT_QPA_FONTDIR=${_toon_test_font_dir}")
        else()
            message(WARNING "Windows offscreen tests need installed fonts; set QT_QPA_FONTDIR to a font directory.")
        endif()
    endif()
endif()

if(MSVC)
    # A portable ZIP must carry its compiler DLLs beside ToonPro.exe. A smoke
    # run on a Visual Studio runner otherwise silently uses the installed CRT.
    # Windows 10/11 already provide UCRT; never redistribute debug runtimes.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
    set(CMAKE_INSTALL_DEBUG_LIBRARIES FALSE)
    set(CMAKE_INSTALL_DEBUG_LIBRARIES_ONLY FALSE)
    set(CMAKE_INSTALL_UCRT_LIBRARIES FALSE)
    include(InstallRequiredSystemLibraries)
    set(_toon_crt_names)
    foreach(_toon_crt IN LISTS CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS)
        if(NOT EXISTS "${_toon_crt}")
            message(FATAL_ERROR "Required MSVC runtime is missing: ${_toon_crt}")
        endif()
        get_filename_component(_toon_crt_name "${_toon_crt}" NAME)
        string(TOLOWER "${_toon_crt_name}" _toon_crt_name)
        list(APPEND _toon_crt_names "${_toon_crt_name}")
    endforeach()
    foreach(_toon_required_crt msvcp140.dll vcruntime140.dll vcruntime140_1.dll)
        if(NOT _toon_required_crt IN_LIST _toon_crt_names)
            message(FATAL_ERROR "Install the Visual Studio 2022 C++ redistributable files before packaging: ${_toon_required_crt} was not found.")
        endif()
    endforeach()
    install(PROGRAMS ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS} DESTINATION ${CMAKE_INSTALL_BINDIR}
        CONFIGURATIONS Release RelWithDebInfo MinSizeRel)
    # Qt's default release deployment adds a separate runtime installer and
    # needs VCINSTALLDIR. App-local deployment above is explicit and verified.
    list(APPEND _toon_deploy_runtime_options NO_COMPILER_RUNTIME)
endif()
